
$exclude = @("node_modules", ".next", ".git", "cache")
$script:output = @()

function Build-Tree($path, $indent = "") {
    Get-ChildItem $path | Where-Object {
        $exclude -notcontains $_.Name
    } | ForEach-Object {

        $script:output += "$indent├── $($_.Name)"

        if ($_.PSIsContainer) {
            Build-Tree $_.FullName "$indent│   "
        }
    }
}
